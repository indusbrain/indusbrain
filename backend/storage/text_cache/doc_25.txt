          using System;

           class Class5                                                             
           {                                                                        
             public bool IsPrime(int n)                                             
             {                                                                      
              if (n <= 1)                                                           
                return false;                                                       
               for (int i = 2; i <= Math.Sqrt(n); i++)                              
               {                                                                    
                if (n % i == 0)                                                     
                  return false;                                                     
               }                                                                    
                                                                                    
               return true;

               var font = richTextBox1.Font ?? this.Font;                           
               var rect = e.MarginBounds;                                           
               e.Graphics.MeasureString(_printText, font, rect.Size, StringFormat.GenericDefault, out
          charsFitted, out linesFilled);                                            
                                                                                    
              // Draw the substring that fits                                       
               string pageText = _printText.Substring(0, charsFitted);              
               e.Graphics.DrawString(pageText, font, Brushes.Black, rect);          
                                                                                    
              // Remove the portion that was printed                                
              if (charsFitted < _printText.Length)                                  
               {                                                                    
                _printText = _printText.Substring(charsFitted);                     
                e.HasMorePages = true;                                              
               }                                                                    
               else                                                                 
               {                                                                    
                _printText = null;                                                  
                e.HasMorePages = false;                                             
               }
